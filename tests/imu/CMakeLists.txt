cmake_minimum_required(VERSION 3.16)
project(imu_tests C CXX)
enable_testing()
add_executable(raise_detector_test ../raise_detector_test.c
  ../../firmware/src/imu/raise_detector.c)
add_executable(raise_trace_replay raise_trace_replay.c ../../firmware/src/imu/raise_detector.c)
add_executable(hbm_switch_test hbm_switch_test.cpp ../../firmware/src/imu/hbm_switch.c)
set_target_properties(hbm_switch_test PROPERTIES CXX_STANDARD 17)
foreach(target raise_detector_test raise_trace_replay hbm_switch_test)
  target_include_directories(${target} PRIVATE ../../firmware/src)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /UNDEBUG /W4)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(${target} PRIVATE m)
  endif()
endforeach()
add_test(NAME raise_detector COMMAND raise_detector_test)
add_test(NAME hbm_switch COMMAND hbm_switch_test)
