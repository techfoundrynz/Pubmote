cmake_minimum_required(VERSION 3.16)
project(garage_game_tests LANGUAGES CXX)
add_executable(garage_game_test game_test.cpp)
target_compile_features(garage_game_test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(garage_game_test PRIVATE /UNDEBUG /W4)
else()
  target_compile_options(garage_game_test PRIVATE -UNDEBUG -Wall -Wextra -pedantic)
endif()
enable_testing()
add_test(NAME garage_game COMMAND garage_game_test)
