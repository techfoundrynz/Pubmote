cmake_minimum_required(VERSION 3.16)
project(radio_host_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()

foreach(test ble wifi diagnostic)
  add_executable(${test}_test ${test}_test.c)
  target_include_directories(${test}_test PRIVATE "${GENERATED_DIR}" fakes "../../firmware/src/utilities")
  if(MSVC)
    target_compile_options(${test}_test PRIVATE /UNDEBUG /experimental:c11atomics)
    target_compile_definitions(${test}_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${test}_test PRIVATE -UNDEBUG -Wall -Wextra)
  endif()
  add_test(NAME ${test} COMMAND ${test}_test)
  set_tests_properties(${test} PROPERTIES TIMEOUT 10)
endforeach()
target_sources(diagnostic_test PRIVATE ../../firmware/src/utilities/log_ring.c)
add_executable(keypad_test keypad_test.cpp)
target_include_directories(keypad_test PRIVATE "../../firmware/src/utilities")
if(MSVC)
  target_compile_options(keypad_test PRIVATE /UNDEBUG)
else()
  target_compile_options(keypad_test PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME keypad COMMAND keypad_test)
