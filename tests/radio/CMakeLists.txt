cmake_minimum_required(VERSION 3.16)
project(radio_host_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()

foreach(test ble wifi diagnostic radio_session espnow ota)
  add_executable(${test}_test ${test}_test.c)
  target_include_directories(${test}_test PRIVATE "${GENERATED_DIR}" fakes
      "../../firmware/modules/support/include"
      "../../firmware/modules/support/include/utilities")
  if(MSVC)
    target_compile_options(${test}_test PRIVATE /UNDEBUG /experimental:c11atomics)
    target_compile_definitions(${test}_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${test}_test PRIVATE -UNDEBUG -Wall -Wextra)
  endif()
  add_test(NAME ${test} COMMAND ${test}_test)
  set_tests_properties(${test} PROPERTIES TIMEOUT 10)
endforeach()
target_sources(diagnostic_test PRIVATE ../../firmware/modules/support/src/log_ring.c)
add_executable(keypad_test keypad_test.cpp)
target_include_directories(keypad_test PRIVATE "../../firmware/modules/support/include/utilities")
if(MSVC)
  target_compile_options(keypad_test PRIVATE /UNDEBUG)
else()
  target_compile_options(keypad_test PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME keypad COMMAND keypad_test)

add_executable(ui_operation_test ui_operation_test.cpp ../../firmware/src/utilities/ui_operation.cpp)
target_include_directories(ui_operation_test PRIVATE "${GENERATED_DIR}" ui_fakes ../../firmware/src)
if(MSVC)
  target_compile_options(ui_operation_test PRIVATE /UNDEBUG)
else()
  target_compile_options(ui_operation_test PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME ui_operation COMMAND ui_operation_test)

add_executable(ota_gate_test ota_gate_test.cpp)
target_include_directories(ota_gate_test PRIVATE "${GENERATED_DIR}")
if(MSVC)
  target_compile_options(ota_gate_test PRIVATE /UNDEBUG)
else()
  target_compile_options(ota_gate_test PRIVATE -UNDEBUG -Wall -Wextra)
endif()
add_test(NAME ota_gate COMMAND ota_gate_test)
